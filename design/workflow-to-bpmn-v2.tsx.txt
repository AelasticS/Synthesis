/** @jsx createExprNode */
import { createExprNode } from '@aelastics/synthesis/jsx'
import type { ExprNode } from '@aelastics/synthesis/jsx'
import { abstractM2M, E2E, SpecPoint, SpecOption } from '@aelastics/synthesis/m2m'
import type { ModelStore } from '@aelastics/synthesis/store'
import * as wm from 'workflow-metamodel/model'
import * as bm from 'bpmn-metamodel/model'
import { Process, Task, StartEvent, EndEvent, Gateway, SubProcess, Flow, FromNode, ToNode } from 'bpmn-metamodel'

/** What an activity contributes to the enclosing graph: nodes and flows. */
type GraphPart = ExprNode<bm.INode> | ExprNode<bm.IFlow>

export class WorkflowToBpmnTransformation extends abstractM2M<wm.IProcess, bm.IProcess> {
  private flowCount = 0
  private parCount = 0
  private parNames = new Map<wm.IParallel, string>()

  constructor(store: ModelStore) {
    super(store)
  }

  template(p: wm.IProcess): ExprNode<wm.IProcess, bm.IProcess> {
    const root = this.rootOf(p)
    return (
      <Process name={`${p.name}_bpmn`} content="" MDA_level="M1" store={this.context.store}>
        <StartEvent name="start" />
        {this.activityToGraph(root)}
        <EndEvent name="end" />
        <Flow name={this.nextFlowName()}>
          <FromNode $refByName="start" />
          {this.toEntryOf(root)}
        </Flow>
        <Flow name={this.nextFlowName()}>
          {this.fromExitOf(root)}
          <ToNode $refByName="end" />
        </Flow>
      </Process>
    )
  }

  // ----- the graph rule: one pure-dispatch point, one option per kind ----

  @E2E()
  @SpecPoint()
  activityToGraph(a: wm.IActivity): GraphPart | GraphPart[] {
    return undefined! // pure dispatcher: the matching option supplies the result
  }

  @SpecOption('activityToGraph', wm.Task)
  taskToGraph(t: wm.ITask): ExprNode<bm.ITask> {
    return <Task name={t.name} />
  }

  @SpecOption('activityToGraph', wm.SubProcess)
  subProcessToGraph(s: wm.ISubProcess): ExprNode<bm.ISubProcess> {
    const body = this.childrenOf(s)
    return (
      <SubProcess name={s.name}>
        <StartEvent name={`${s.name}_start`} />
        {body.flatMap((a) => this.activityToGraph(a))}
        <EndEvent name={`${s.name}_end`} />
        {this.chainFlows(body)}
        <Flow name={this.nextFlowName()}>
          <FromNode $refByName={`${s.name}_start`} />
          {this.toEntryOf(body[0])}
        </Flow>
        <Flow name={this.nextFlowName()}>
          {this.fromExitOf(body[body.length - 1])}
          <ToNode $refByName={`${s.name}_end`} />
        </Flow>
      </SubProcess>
    )
  }

  @SpecOption('activityToGraph', wm.Sequence)
  sequenceToGraph(q: wm.ISequence): GraphPart[] {
    const children = this.childrenOf(q)
    return [...children.flatMap((a) => this.activityToGraph(a)), ...this.chainFlows(children)]
  }

  @SpecOption('activityToGraph', wm.Parallel)
  parallelToGraph(p: wm.IParallel): GraphPart[] {
    const base = this.baseOf(p)
    const children = this.childrenOf(p)
    return [
      <Gateway name={`${base}_split`} kind="parallel" />,
      ...children.flatMap((a) => this.activityToGraph(a)),
      <Gateway name={`${base}_join`} kind="parallel" />,
      ...children.map((c) => (
        <Flow name={this.nextFlowName()}>
          <FromNode $refByName={`${base}_split`} />
          {this.toEntryOf(c)}
        </Flow>
      )),
      ...children.map((c) => (
        <Flow name={this.nextFlowName()}>
          {this.fromExitOf(c)}
          <ToNode $refByName={`${base}_join`} />
        </Flow>
      )),
    ]
  }

  // ----- boundary references: where a block's graph begins and ends ------
  // An entry is only ever a flow target, an exit only ever a flow source,
  // so the references come back already wearing the right tag.

  @SpecPoint()
  toEntryOf(a: wm.IActivity): ExprNode<bm.INode> {
    return undefined! // pure dispatcher
  }

  @SpecOption('toEntryOf', wm.Task)
  toEntryOfTask(t: wm.ITask): ExprNode<bm.INode> {
    return <ToNode $refByName={t.name} />
  }

  @SpecOption('toEntryOf', wm.SubProcess)
  toEntryOfSubProcess(s: wm.ISubProcess): ExprNode<bm.INode> {
    return <ToNode $refByName={s.name} />
  }

  @SpecOption('toEntryOf', wm.Sequence)
  toEntryOfSequence(q: wm.ISequence): ExprNode<bm.INode> {
    return this.toEntryOf(this.childrenOf(q)[0])
  }

  @SpecOption('toEntryOf', wm.Parallel)
  toEntryOfParallel(p: wm.IParallel): ExprNode<bm.INode> {
    return <ToNode $refByName={`${this.baseOf(p)}_split`} />
  }

  @SpecPoint()
  fromExitOf(a: wm.IActivity): ExprNode<bm.INode> {
    return undefined! // pure dispatcher
  }

  @SpecOption('fromExitOf', wm.Task)
  fromExitOfTask(t: wm.ITask): ExprNode<bm.INode> {
    return <FromNode $refByName={t.name} />
  }

  @SpecOption('fromExitOf', wm.SubProcess)
  fromExitOfSubProcess(s: wm.ISubProcess): ExprNode<bm.INode> {
    return <FromNode $refByName={s.name} />
  }

  @SpecOption('fromExitOf', wm.Sequence)
  fromExitOfSequence(q: wm.ISequence): ExprNode<bm.INode> {
    const children = this.childrenOf(q)
    return this.fromExitOf(children[children.length - 1])
  }

  @SpecOption('fromExitOf', wm.Parallel)
  fromExitOfParallel(p: wm.IParallel): ExprNode<bm.INode> {
    return <FromNode $refByName={`${this.baseOf(p)}_join`} />
  }

  // ----- small helpers ---------------------------------------------------

  private chainFlows(children: wm.IActivity[]): ExprNode<bm.IFlow>[] {
    const flows: ExprNode<bm.IFlow>[] = []
    for (let i = 0; i < children.length - 1; i++) {
      flows.push(
        <Flow name={this.nextFlowName()}>
          {this.fromExitOf(children[i])}
          {this.toEntryOf(children[i + 1])}
        </Flow>
      )
    }
    return flows
  }

  private nextFlowName(): string {
    return `f${++this.flowCount}`
  }

  // Stable base name for a parallel's gateway pair: its own name, or a
  // generated one remembered per element, so the graph rule and the
  // boundary references always agree.
  private baseOf(p: wm.IParallel): string {
    if (p.name) return p.name
    let base = this.parNames.get(p)
    if (base === undefined) {
      base = `par${++this.parCount}`
      this.parNames.set(p, base)
    }
    return base
  }

  private childrenOf(a: wm.IActivity): wm.IActivity[] {
    const children = (a.children ?? []) as wm.IActivity[]
    if (children.length === 0) {
      throw new Error(`Block '${a.name}' is empty: it has no entry or exit`)
    }
    return children
  }

  private rootOf(p: wm.IProcess): wm.IActivity {
    if (!p.body) {
      throw new Error(`Process '${p.name}' has no body block`)
    }
    return p.body as wm.IActivity
  }
}
